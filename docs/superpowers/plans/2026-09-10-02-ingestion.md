# Swiss Grounding MCP — Plan 2: Ingestion, change detection, source suggestions

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the fixture loader with a real, re-runnable ingestion pipeline: fetch from Fedlex, HTML sites, PDFs and JSON APIs; detect changes; translate and tag; narrow extent; chunk; embed; persist with versioning; and let MCP clients suggest new sources.

**Architecture:** A batch pipeline (`ingest/pipeline.py`) runs one source through pure stages: fetch (per-strategy fetcher) → change detection → enrichment (translation + category tags via an LLM client with a fake backend) → extent narrowing → chunking → embedding → persistence (versioned documents, chunk-level diffs). Fetchers return immutable `FetchedDocument`s and never touch the database; persistence is the only stage that writes. Source suggestions are a registry concern with domain classification and auto-approval for official domains.

**Tech Stack:** Python 3.12, httpx, beautifulsoup4 + markdownify (HTML), pypdf (PDF), anthropic SDK (translation and tagging; `fake` backend for tests), existing psycopg/pgvector stack, typer, FastMCP (mcp 1.x).

**Spec:** `docs/superpowers/specs/2026-09-10-swiss-grounding-mcp-design.md` sections 4 (`source_suggestion`, `document` versioning, `document_change`), 5, 7 (`swiss_suggest_source`, `swiss_coverage` suggestions), 8 (`ingest`, `sources`), 10.

```python
import hashlib

from swiss_grounding.ingest.models import FetchedDocument


def text_hash(text: str) -> str:
    return hashlib.sha256(text.encode()).hexdigest()


def content_hash(doc: FetchedDocument) -> str:
    """Hash of the original section texts only; title and metadata changes do not count."""
    return text_hash("\n␞".join(s.text for s in doc.sections))
```

```python
# src/swiss_grounding/ingest/html_sections.py
from dataclasses import dataclass
from urllib.parse import urljoin, urldefrag

from bs4 import BeautifulSoup, Tag
from markdownify import markdownify

from swiss_grounding.ingest.models import Section

DROP_TAGS = ("nav", "footer", "header", "script", "style", "noscript", "aside", "form")
HEADING_TAGS = ("h1", "h2", "h3", "h4")
MAIN_CANDIDATES = ("main", "article", "[role=main]", "body")


@dataclass(frozen=True)
class Page:
    title: str
    language: str
    sections: tuple[Section, ...]


def _content_root(soup: BeautifulSoup) -> Tag | None:
    for selector in MAIN_CANDIDATES:
        node = soup.select_one(selector)
        if node is not None:
            return node
    return None


def _clean(root: Tag) -> Tag:
    for tag in root.find_all(DROP_TAGS):
        tag.decompose()
    return root


def _title(soup: BeautifulSoup) -> str:
    h1 = soup.find("h1")
    if h1 and h1.get_text(strip=True):
        return h1.get_text(" ", strip=True)
    if soup.title and soup.title.string:
        return soup.title.string.split("|")[0].strip()
    return ""


def _language(soup: BeautifulSoup, default: str) -> str:
    html = soup.find("html")
    lang = html.get("lang") if html else None
    return lang.split("-")[0].lower() if lang else default


def _to_text(nodes: list[Tag]) -> str:
    html = "".join(str(n) for n in nodes)
    return markdownify(html, heading_style="ATX", strip=["a", "img"]).strip()


def _split_by_headings(root: Tag, title: str) -> tuple[Section, ...]:
    sections: list[Section] = []
    stack: list[tuple[int, str]] = []
    anchor: str | None = None
    buffer: list[Tag] = []
    heading_seen = False

    def flush() -> None:
        text = _to_text(buffer)
        if text:
            path = " > ".join(name for _, name in stack) or title
            sections.append(Section(heading_path=path, anchor=anchor, text=text))
        buffer.clear()

    for node in root.find_all(recursive=False):
        if node.name in HEADING_TAGS:
            flush()
            heading_seen = True
            level = int(node.name[1])
            name = node.get_text(" ", strip=True)
            while stack and stack[-1][0] >= level:
                stack.pop()
            stack.append((level, name))
            anchor = node.get("id")
        else:
            buffer.append(node)
    flush()
    if not heading_seen and not sections:
        return ()
    return tuple(sections)


def html_to_page(html: str, url: str, default_language: str) -> Page:
    """Parse an HTML page into a title, a language, and heading-delimited sections."""
    soup = BeautifulSoup(html, "lxml")
    title = _title(soup)
    language = _language(soup, default_language)
    root = _content_root(soup)
    if root is None:
        return Page(title=title, language=language, sections=())
    sections = _split_by_headings(_clean(root), title)
    return Page(title=title, language=language, sections=sections)


def extract_links(html: str, base_url: str) -> tuple[str, ...]:
    """All absolute http(s) links on the page, fragments removed, order preserved, deduplicated."""
    soup = BeautifulSoup(html, "lxml")
    seen: dict[str, None] = {}
    for a in soup.find_all("a", href=True):
        absolute, _ = urldefrag(urljoin(base_url, a["href"]))
        if absolute.startswith(("http://", "https://")):
            seen.setdefault(absolute, None)
    return tuple(seen)
```

Note for the engineer: `_split_by_headings` iterates the direct children of the content root. If a fixture nests headings inside wrappers (`<div>`), the test `test_sections_follow_headings_with_paths_and_anchors` will fail; in that case change the iteration to `root.descendants`-based walking that treats heading tags as boundaries and everything else as content, and keep the tests unchanged. The fixture above has headings as direct children of `<main>`.

```python
def source(**overrides) -> Source:

def make(handler, known=None, limits=CrawlLimits(max_pages=10, delay_seconds=0)):
    http = httpx.Client(transport=httpx.MockTransport(handler), headers={"user-agent": "t"})
    return HtmlCrawlFetcher(http=http, limits=limits), (known or {})


def routes(request, calls):
    calls.append((request.method, str(request.url), request.headers.get("if-none-match")))
    path = request.url.path
    if path == "/robots.txt":
        return httpx.Response(200, text=ROBOTS)
    if path == "/de/tiere/a.html":
        return httpx.Response(200, text=PAGE_A, headers={"etag": '"a1"', "last-modified": "Mon, 01 Sep 2026 00:00:00 GMT"})
    if path == "/de/tiere/b.html":
        if request.headers.get("if-none-match") == '"b1"':
            return httpx.Response(304)
        return httpx.Response(200, text=PAGE_B, headers={"etag": '"b1"'})
    if path == "/de/tiere/gone.html":
        return httpx.Response(404)
    return httpx.Response(404)
```

```python
# src/swiss_grounding/ingest/fetchers/html_crawl.py
import time
from collections import deque
from dataclasses import dataclass
from typing import Mapping
from urllib import robotparser
from urllib.parse import urljoin, urlparse

import httpx

from swiss_grounding.ingest.html_sections import extract_links, html_to_page
from swiss_grounding.ingest.models import FetchedDocument, FetchError, FetchResult, KnownUrl
from swiss_grounding.registry.models import Source

SKIP_SUFFIXES = (".pdf", ".jpg", ".jpeg", ".png", ".gif", ".svg", ".zip", ".xml", ".json", ".ics", ".csv")


@dataclass(frozen=True)
class CrawlLimits:
    max_pages: int = 200
    delay_seconds: float = 0.5


@dataclass(frozen=True)
class _Fetched:
    doc: FetchedDocument | None
    links: tuple[str, ...]
    status: int


class HtmlCrawlFetcher:
    """Bounded same-domain crawl with path allowlist, robots.txt, and conditional requests."""

    def __init__(self, http: httpx.Client, limits: CrawlLimits):
        self._http = http
        self._limits = limits

    def fetch(self, source: Source, known: Mapping[str, KnownUrl]) -> FetchResult:
        base = urlparse(source.base_url)
        allow = tuple(source.fetch_config.get("path_allowlist", ["/"]))
        starts = tuple(source.fetch_config.get("start_paths", allow))
        robots = self._robots(source.base_url)
        queue = deque(urljoin(source.base_url, p) for p in starts)
        queue.extend(u for u in known if u not in queue)
        seen: set[str] = set()
        documents: list[FetchedDocument] = []
        unchanged: list[str] = []
        gone: list[str] = []
        errors: list[FetchError] = []
        pages = 0
        while queue and pages < self._limits.max_pages:
            url = queue.popleft()
            if url in seen or not self._allowed(url, base, allow, robots):
                continue
            seen.add(url)
            pages += 1
            try:
                fetched = self._get(url, source, known.get(url))
            except httpx.HTTPError as exc:
                errors.append(FetchError(url=url, reason=str(exc)))
                continue
            if fetched.status == 304:
                unchanged.append(url)
            elif fetched.status in (404, 410):
                if url in known:
                    gone.append(url)
            elif fetched.doc is not None:
                documents.append(fetched.doc)
            queue.extend(link for link in fetched.links if link not in seen)
            if self._limits.delay_seconds:
                time.sleep(self._limits.delay_seconds)
        return FetchResult(documents=tuple(documents), unchanged_urls=tuple(unchanged),
                           gone_urls=tuple(gone), errors=tuple(errors))

    def _robots(self, base_url: str) -> robotparser.RobotFileParser:
        parser = robotparser.RobotFileParser()
        try:
            response = self._http.get(urljoin(base_url, "/robots.txt"))
            parser.parse(response.text.splitlines() if response.status_code == 200 else [])
        except httpx.HTTPError:
            parser.parse([])
        return parser

    def _allowed(self, url: str, base, allow: tuple[str, ...], robots) -> bool:
        parsed = urlparse(url)
        if parsed.netloc != base.netloc or parsed.scheme not in ("http", "https"):
            return False
        if parsed.path.lower().endswith(SKIP_SUFFIXES):
            return False
        if not any(parsed.path.startswith(prefix) for prefix in allow):
            return False
        return robots.can_fetch("*", url)

    def _get(self, url: str, source: Source, prior: KnownUrl | None) -> _Fetched:
        headers = {}
        if prior and prior.etag:
            headers["if-none-match"] = prior.etag
        if prior and prior.last_modified:
            headers["if-modified-since"] = prior.last_modified
        response = self._http.get(url, headers=headers, follow_redirects=True)
        if response.status_code == 304:
            return _Fetched(doc=None, links=(), status=304)
        if response.status_code >= 400:
            return _Fetched(doc=None, links=(), status=response.status_code)
        if "html" not in response.headers.get("content-type", "text/html"):
            return _Fetched(doc=None, links=(), status=response.status_code)
        page = html_to_page(response.text, url=url, default_language=source.fetch_config.get("language", "de"))
        links = extract_links(response.text, base_url=url)
        if not page.sections:
            return _Fetched(doc=None, links=links, status=response.status_code)
        doc = FetchedDocument(
            source_id=source.id, url=url, title=page.title or url, language=page.language,
            sections=page.sections, etag=response.headers.get("etag"),
            last_modified=response.headers.get("last-modified"),
        )
        return _Fetched(doc=doc, links=links, status=response.status_code)
```

```python
# src/swiss_grounding/ingest/fetchers/pdf.py
import io
from typing import Callable, Mapping

import httpx
from pypdf import PdfReader

from swiss_grounding.ingest.models import FetchedDocument, FetchError, FetchResult, KnownUrl, Section
from swiss_grounding.registry.models import Source

PageExtractor = Callable[[bytes], tuple[str, ...]]


def extract_pages_with_pypdf(data: bytes) -> tuple[str, ...]:
    reader = PdfReader(io.BytesIO(data))
    return tuple((page.extract_text() or "").strip() for page in reader.pages)


class PdfFetcher:
    """Fetch configured PDF URLs; HEAD first for change detection, then download and split by page."""

    def __init__(self, http: httpx.Client, extract_pages: PageExtractor = extract_pages_with_pypdf):
        self._http = http
        self._extract = extract_pages

    def fetch(self, source: Source, known: Mapping[str, KnownUrl]) -> FetchResult:
        urls = tuple(source.fetch_config.get("urls", ()))
        language = source.fetch_config.get("language", "de")
        documents, unchanged, gone, errors = [], [], [], []
        for url in urls:
            try:
                head = self._http.head(url, follow_redirects=True)
                if head.status_code in (404, 410):
                    if url in known:
                        gone.append(url)
                    continue
                last_modified = head.headers.get("last-modified")
                prior = known.get(url)
                if prior and last_modified and prior.last_modified == last_modified:
                    unchanged.append(url)
                    continue
                response = self._http.get(url, follow_redirects=True)
                response.raise_for_status()
                pages = self._extract(response.content)
            except (httpx.HTTPError, ValueError) as exc:
                errors.append(FetchError(url=url, reason=str(exc)))
                continue
            sections = tuple(
                Section(heading_path=f"Page {i}", anchor=f"page={i}", text=text)
                for i, text in enumerate(pages, 1) if text
            )
            if not sections:
                errors.append(FetchError(url=url, reason="no extractable text"))
                continue
            documents.append(FetchedDocument(
                source_id=source.id, url=url, title=url.rsplit("/", 1)[-1], language=language,
                sections=sections, etag=head.headers.get("etag"), last_modified=last_modified,
            ))
        return FetchResult(documents=tuple(documents), unchanged_urls=tuple(unchanged),
                           gone_urls=tuple(gone), errors=tuple(errors))
```

```python
# src/swiss_grounding/ingest/fetchers/api.py
from typing import Any, Mapping

import httpx

from swiss_grounding.ingest.models import FetchedDocument, FetchError, FetchResult, KnownUrl, Section
from swiss_grounding.registry.models import Source


def _dig(payload: Any, dotted: str) -> Any:
    node = payload
    for key in dotted.split("."):
        if not isinstance(node, dict) or key not in node:
            raise ValueError(f"records_path {dotted!r} not found in payload")
        node = node[key]
    if not isinstance(node, list):
        raise ValueError(f"records_path {dotted!r} does not point to a list")
    return node


class ApiFetcher:
    """Turn each record of a JSON feed into a small document, keyed by feed URL + record id."""

    def __init__(self, http: httpx.Client):
        self._http = http

    def fetch(self, source: Source, known: Mapping[str, KnownUrl]) -> FetchResult:
        cfg = source.fetch_config
        feed_url = cfg["url"]
        try:
            response = self._http.get(feed_url)
            response.raise_for_status()
            records = _dig(response.json(), cfg.get("records_path", ""))
        except (httpx.HTTPError, ValueError, KeyError) as exc:
            return FetchResult(documents=(), errors=(FetchError(url=feed_url, reason=str(exc)),))
        documents = []
        for record in records:
            record_id = str(record.get(cfg.get("id_field", "id"), ""))
            if not record_id:
                continue
            url = f"{feed_url}#{record_id}"
            documents.append(FetchedDocument(
                source_id=source.id, url=url, identifier=record_id,
                title=cfg["title_template"].format_map(record), language=cfg.get("language", "en"),
                sections=(Section(heading_path=cfg["title_template"].format_map(record), anchor=None,
                                  text=cfg["text_template"].format_map(record).strip()),),
            ))
        present = {d.url for d in documents}
        gone = tuple(u for u in known if u.startswith(feed_url + "#") and u not in present)
        return FetchResult(documents=tuple(documents), gone_urls=gone)
```

```python
from datetime import date
from typing import Any, Mapping

import httpx
from bs4 import BeautifulSoup

from swiss_grounding.ingest.models import FetchedDocument, FetchError, FetchResult, KnownUrl, Section
from swiss_grounding.registry.models import Source

SPARQL_ENDPOINT = "https://fedlex.data.admin.ch/sparqlendpoint"
LANGUAGE_URI = {
    "de": "http://publications.europa.eu/resource/authority/language/DEU",
    "fr": "http://publications.europa.eu/resource/authority/language/FRA",
    "it": "http://publications.europa.eu/resource/authority/language/ITA",
}
XML_FORMAT_URI = "http://publications.europa.eu/resource/authority/file-type/XML"

# Current consolidation of one SR number in one language, with the Akoma Ntoso file URL.
QUERY = """
PREFIX jolux: <http://data.legilux.public.lu/resource/ontology/jolux#>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>
SELECT ?abstract ?consolidation ?dateApplicability ?dateEndApplicability ?title ?fileUrl WHERE {
  ?abstract a jolux:ConsolidationAbstract ;
            jolux:classifiedByTaxonomyEntry ?entry .
  ?entry skos:notation ?sr .
  FILTER(str(?sr) = "%(sr)s")
  ?consolidation jolux:isMemberOf ?abstract ;
                 jolux:dateApplicability ?dateApplicability ;
                 jolux:isRealizedBy ?expr .
  OPTIONAL { ?consolidation jolux:dateEndApplicability ?dateEndApplicability }
  ?expr jolux:language <%(lang)s> ;
        jolux:isEmbodiedBy ?manif .
  OPTIONAL { ?expr jolux:title ?title }
  ?manif jolux:format <%(xml)s> ;
         jolux:isExemplifiedBy ?fileUrl .
  FILTER(!bound(?dateEndApplicability) || ?dateEndApplicability >= now())
}
ORDER BY DESC(?dateApplicability)
LIMIT 1
"""


def parse_akoma_ntoso(xml: str) -> tuple[Section, ...]:
    """One section per <article>: heading_path 'Art. N Heading', anchor = eId, text = paragraphs."""
    soup = BeautifulSoup(xml, "xml")
    sections = []
    for article in soup.find_all("article"):
        num = article.find("num", recursive=False)
        heading = article.find("heading", recursive=False)
        label = " ".join(x.get_text(" ", strip=True) for x in (num, heading) if x)
        paragraphs = [
            p.get_text(" ", strip=True)
            for p in article.find_all("content")
        ]
        text = "\n".join(t for t in paragraphs if t)
        if text:
            sections.append(Section(heading_path=label or article.get("eId", ""), anchor=article.get("eId"), text=text))
    return tuple(sections)


def _value(binding: dict[str, Any], key: str) -> str | None:
    node = binding.get(key)
    return node["value"] if node else None


def _page_url(abstract_uri: str, date_applicability: str, lang: str) -> str:
    # https://fedlex.data.admin.ch/eli/cc/2008/414 -> https://www.fedlex.admin.ch/eli/cc/2008/414/20220101/de
    path = abstract_uri.split("fedlex.data.admin.ch", 1)[1]
    return f"https://www.fedlex.admin.ch{path}/{date_applicability.replace('-', '')}/{lang}"


class FedlexFetcher:
    """Resolve SR numbers to their current consolidation and parse the Akoma Ntoso XML."""

    def __init__(self, http: httpx.Client, endpoint: str = SPARQL_ENDPOINT):
        self._http = http
        self._endpoint = endpoint

    def fetch(self, source: Source, known: Mapping[str, KnownUrl]) -> FetchResult:
        lang = source.fetch_config.get("language", "de")
        documents, unchanged, errors = [], [], []
        for sr in source.fetch_config.get("sr_numbers", ()):
            try:
                binding = self._current(sr, lang)
            except (httpx.HTTPError, ValueError) as exc:
                errors.append(FetchError(url=self._endpoint, reason=f"SR {sr}: {exc}"))
                continue
            if binding is None:
                errors.append(FetchError(url=self._endpoint, reason=f"SR {sr}: no current consolidation"))
                continue
            date_app = _value(binding, "dateApplicability")
            url = _page_url(_value(binding, "abstract"), date_app, lang)
            if url in known:
                unchanged.append(url)
                continue
            try:
                xml = self._http.get(_value(binding, "fileUrl"), follow_redirects=True)
                xml.raise_for_status()
                sections = parse_akoma_ntoso(xml.text)
            except httpx.HTTPError as exc:
                errors.append(FetchError(url=url, reason=str(exc)))
                continue
            if not sections:
                errors.append(FetchError(url=url, reason="no articles parsed"))
                continue
            end = _value(binding, "dateEndApplicability")
            documents.append(FetchedDocument(
                source_id=source.id, url=url, identifier=f"SR {sr}",
                title=_value(binding, "title") or f"SR {sr}", language=lang, sections=sections,
                valid_from=date.fromisoformat(date_app), valid_to=date.fromisoformat(end) if end else None,
                in_force=end is None,
            ))
        return FetchResult(documents=tuple(documents), unchanged_urls=tuple(unchanged), errors=tuple(errors))

    def _current(self, sr: str, lang: str) -> dict[str, Any] | None:
        query = QUERY % {"sr": sr, "lang": LANGUAGE_URI[lang], "xml": XML_FORMAT_URI}
        response = self._http.post(
            self._endpoint, data={"query": query},
            headers={"accept": "application/sparql-results+json"},
        )
        response.raise_for_status()
        bindings = response.json().get("results", {}).get("bindings", [])
        return bindings[0] if bindings else None
```

```python
# src/swiss_grounding/ingest/llm.py
import json
from typing import Any, Protocol, Sequence

from swiss_grounding.ingest.models import CategoryTag
from swiss_grounding.registry.categories import Category

MAX_TOKENS = 4096
FAKE_KEYWORD_CONFIDENCE = 0.9

TRANSLATE_PROMPT = (
    "Translate the following {language} text to English. Preserve paragraph breaks, numbering, "
    "article references and proper nouns. Output only the translation.\n\n{text}"
)
CLASSIFY_PROMPT = (
    "Classify the document below into zero or more of these categories. "
    "Reply with JSON only: {{\"categories\": [{{\"id\": \"...\", \"confidence\": 0.0-1.0}}]}}.\n"
    "Categories:\n{categories}\n\nTitle: {title}\n\nText:\n{text}"
)


class LLM(Protocol):
    def translate(self, text: str, source_language: str) -> str: ...
    def classify(self, title: str, text: str, categories: Sequence[Category]) -> tuple[CategoryTag, ...]: ...


def parse_tags(raw: str, known_ids: Sequence[str]) -> tuple[CategoryTag, ...]:
    """Parse the JSON tag response; unknown ids dropped, confidence clamped to [0, 1]."""
    try:
        data = json.loads(raw)
        items = data["categories"]
    except (ValueError, KeyError, TypeError) as exc:
        raise ValueError(f"unparseable tag response: {raw[:200]!r}") from exc
    tags = []
    for item in items:
        cid = item.get("id")
        if cid in known_ids:
            confidence = min(1.0, max(0.0, float(item.get("confidence", 0.0))))
            tags.append(CategoryTag(id=cid, confidence=confidence))
    return tuple(tags)


class FakeLLM:
    """Deterministic stand-in: identity translation, keyword-based tags. Tests and offline demos."""

    def translate(self, text: str, source_language: str) -> str:
        return text

    def classify(self, title: str, text: str, categories: Sequence[Category]) -> tuple[CategoryTag, ...]:
        haystack = f"{title} {text}".lower()
        tags = []
        for c in categories:
            keywords = {c.id, *c.label.lower().split(), *c.description.lower().replace(",", " ").split()}
            keywords = {k for k in keywords if len(k) > 3}
            if any(k in haystack for k in keywords):
                tags.append(CategoryTag(id=c.id, confidence=FAKE_KEYWORD_CONFIDENCE))
        return tuple(tags)


class AnthropicLLM:
    def __init__(self, model: str, client: Any):
        self._model = model
        self._client = client

    @classmethod
    def from_api_key(cls, model: str, api_key: str) -> "AnthropicLLM":
        from anthropic import Anthropic

        return cls(model=model, client=Anthropic(api_key=api_key))

    def _ask(self, prompt: str) -> str:
        message = self._client.messages.create(
            model=self._model, max_tokens=MAX_TOKENS,
            messages=[{"role": "user", "content": prompt}],
        )
        return "".join(getattr(block, "text", "") for block in message.content).strip()

    def translate(self, text: str, source_language: str) -> str:
        return self._ask(TRANSLATE_PROMPT.format(language=source_language, text=text))

    def classify(self, title: str, text: str, categories: Sequence[Category]) -> tuple[CategoryTag, ...]:
        listing = "\n".join(f"- {c.id}: {c.label} ({c.description})" for c in categories)
        raw = self._ask(CLASSIFY_PROMPT.format(categories=listing, title=title, text=text[:6000]))
        return parse_tags(raw, tuple(c.id for c in categories))


def build_llm(backend: str, model: str, api_key: str | None) -> LLM:
    if backend == "fake":
        return FakeLLM()
    if backend == "anthropic":
        if not api_key:
            raise ValueError("anthropic backend requires an api key")
        return AnthropicLLM.from_api_key(model=model, api_key=api_key)
    raise ValueError(f"unknown llm backend: {backend}")
```

```python
# src/swiss_grounding/ingest/narrowing.py
import re

from swiss_grounding.locations.tree import LocationTree, normalize

WORD_RE = re.compile(r"[A-Za-zÀ-ÿ][A-Za-zÀ-ÿ\-]*")
MIN_NAME_LEN = 3
MAX_NGRAM = 3


def _candidate_names(tree: LocationTree) -> dict[str, str]:
    """normalized name -> location id, for names that are safe to match in prose."""
    names: dict[str, str] = {}
    for loc in tree.all():
        for name in loc.all_names():
            if len(name) < MIN_NAME_LEN or (len(name) == 2 and name.isupper()):
                continue
            if not name[:1].isupper() and not name.isdigit():
                continue
            names[normalize(name)] = loc.id
    return names


def _mentions(tree: LocationTree, text: str) -> tuple[str, ...]:
    names = _candidate_names(tree)
    words = WORD_RE.findall(text)
    found: list[str] = []
    for size in range(MAX_NGRAM, 0, -1):
        for start in range(len(words) - size + 1):
            span = words[start:start + size]
            if size == 1 and not span[0][:1].isupper():
                continue
            key = normalize(" ".join(span))
            if key in names:
                found.append(names[key])
    return tuple(dict.fromkeys(found))


def _deepest_common_ancestor(tree: LocationTree, ids: tuple[str, ...]) -> str:
    paths = [tree.path(i) for i in ids]
    common = paths[0]
    for p in paths[1:]:
        common = tuple(a for a, b in zip(common, p) if a == b)
    return common[-1]


def narrow_extent(tree: LocationTree, default_extent_id: str, text: str) -> tuple[str, str]:
    """Narrow to a descendant of the default extent when every mention lies under one; never widen."""
    default_path = tree.path(default_extent_id)
    mentions = tuple(m for m in _mentions(tree, text) if default_extent_id in tree.path(m) and m != default_extent_id)
    if not mentions:
        return default_extent_id, "source_default"
    target = _deepest_common_ancestor(tree, mentions)
    if target == default_extent_id or default_extent_id not in tree.path(target):
        return default_extent_id, "source_default"
    labels = ",".join(tree.get(m).label("en") for m in mentions)
    return target, f"narrowed:{labels}"
```

Note: `test_keeps_default_when_mentions_span_siblings` relies on "Basel" (CH-BS) and "Geneva" (CH-GE) having deepest common ancestor `CH`, which equals the default, so no narrowing. `test_never_widens` relies on filtering mentions to descendants of the default. `_deepest_common_ancestor` also protects against a mention like "Zurich" resolving to the canton alias when the default is the canton itself (`m != default_extent_id`).

```python
# src/swiss_grounding/ingest/chunking.py
from dataclasses import dataclass

from swiss_grounding.ingest.models import ChunkDraft, Section


@dataclass(frozen=True)
class ChunkLimits:
    max_chars: int = 1500
    overlap_chars: int = 150


def _split(text: str, limits: ChunkLimits) -> tuple[str, ...]:
    """Split on sentence-ish boundaries so pieces stay under max_chars, with a small overlap."""
    if len(text) <= limits.max_chars:
        return (text,)
    pieces: list[str] = []
    start = 0
    while start < len(text):
        end = min(len(text), start + limits.max_chars)
        if end < len(text):
            cut = text.rfind(". ", start, end)
            if cut > start + limits.max_chars // 2:
                end = cut + 1
        pieces.append(text[start:end].strip())
        if end >= len(text):
            break
        start = max(end - limits.overlap_chars, start + 1)
    return tuple(p for p in pieces if p)


def chunk_sections(
    original: tuple[Section, ...], english: tuple[Section, ...], limits: ChunkLimits,
) -> tuple[ChunkDraft, ...]:
    """Structure-aware chunking: one chunk per section, split only when the English exceeds the cap.

    The original-language text is attached whole to every piece of a split section, because the
    citation quotes the original and article boundaries are what matter there.
    """
    if len(original) != len(english):
        raise ValueError(f"section count mismatch: {len(original)} original vs {len(english)} english")
    drafts: list[ChunkDraft] = []
    for orig, en in zip(original, english):
        if not en.text.strip():
            continue
        for piece in _split(en.text.strip(), limits):
            drafts.append(ChunkDraft(
                ordinal=len(drafts), heading_path=en.heading_path, anchor=en.anchor,
                text_original=orig.text.strip(), text_en=piece,
            ))
    return tuple(drafts)
```

```python
from typing import Sequence

from swiss_grounding.ingest.llm import LLM
from swiss_grounding.ingest.models import CategoryTag, EnrichedDocument, FetchedDocument, Section
from swiss_grounding.ingest.narrowing import narrow_extent
from swiss_grounding.locations.tree import LocationTree
from swiss_grounding.registry.categories import Category
from swiss_grounding.registry.models import Source

SOURCE_DEFAULT_CONFIDENCE = 1.0
SECTION_SEPARATOR = "\n\n"


def _translate_sections(doc: FetchedDocument, llm: LLM) -> tuple[Section, ...]:
    if doc.language == "en":
        return doc.sections
    return tuple(
        Section(heading_path=s.heading_path, anchor=s.anchor, text=llm.translate(s.text, doc.language))
        for s in doc.sections
    )


def _merge_tags(defaults: Sequence[str], llm_tags: Sequence[CategoryTag]) -> tuple[CategoryTag, ...]:
    merged = {t.id: t for t in llm_tags}
    for cid in defaults:
        merged[cid] = CategoryTag(id=cid, confidence=SOURCE_DEFAULT_CONFIDENCE)
    return tuple(sorted(merged.values(), key=lambda t: (-t.confidence, t.id)))


def enrich(
    doc: FetchedDocument, source: Source, tree: LocationTree, llm: LLM, categories: Sequence[Category],
) -> EnrichedDocument:
    """Translate to English, tag categories, and narrow the extent when the source allows it."""
    sections_en = _translate_sections(doc, llm)
    title_en = doc.title if doc.language == "en" else llm.translate(doc.title, doc.language)
    text_en = SECTION_SEPARATOR.join(s.text for s in sections_en)
    tags = _merge_tags(source.categories, llm.classify(title_en, text_en, categories))
    if source.allow_narrowing:
        extent_id, reason = narrow_extent(tree, source.default_extent_id, text_en)
    else:
        extent_id, reason = source.default_extent_id, "source_default"
    return EnrichedDocument(fetched=doc, title_en=title_en, sections_en=sections_en,
                            extent_id=extent_id, extent_reason=reason, categories=tags)
```

```python
import psycopg

from swiss_grounding.ingest.models import KnownUrl

UPSERT = """
insert into fetch_state (url, source_id, etag, last_modified, content_hash, last_seen_at)
values (%(url)s, %(source_id)s, %(etag)s, %(last_modified)s, %(content_hash)s, now())
on conflict (url) do update set source_id = excluded.source_id, etag = excluded.etag,
    last_modified = excluded.last_modified, content_hash = excluded.content_hash, last_seen_at = now()
"""


def load_known(conn: psycopg.Connection, source_id: str) -> dict[str, KnownUrl]:
    rows = conn.execute(
        "select url, etag, last_modified, content_hash from fetch_state where source_id = %s", (source_id,)
    ).fetchall()
    return {r["url"]: KnownUrl(etag=r["etag"], last_modified=r["last_modified"], content_hash=r["content_hash"])
            for r in rows}


def save_known(conn: psycopg.Connection, source_id: str, url: str, known: KnownUrl) -> None:
    conn.execute(UPSERT, {"url": url, "source_id": source_id, "etag": known.etag,
                          "last_modified": known.last_modified, "content_hash": known.content_hash})


def touch_known(conn: psycopg.Connection, url: str) -> None:
    conn.execute("update fetch_state set last_seen_at = now() where url = %s", (url,))


def forget_known(conn: psycopg.Connection, url: str) -> None:
    conn.execute("delete from fetch_state where url = %s", (url,))
```

```python
@dataclass(frozen=True)
class PersistOutcome:
    document_id: int | None
    change_type: str | None      # "added" | "modified" | None (unchanged)


def _current(conn: psycopg.Connection, url: str) -> dict | None:

        "select id, version, content_hash from document where url = %s and status = 'current'", (url,)
    ).fetchone()


def _insert_document(conn: psycopg.Connection, doc: EnrichedDocument, chash: str, version: int) -> int:
    f = doc.fetched
    row = conn.execute(
        """
        insert into document (source_id, url, identifier, title_original, title_en, language, extent_id,
                              extent_reason, authority, categories, valid_from, valid_to, in_force,
                              published_at, content_hash, version)
        select %s, %s, %s, %s, %s, %s, %s, %s, s.authority, %s::jsonb, %s, %s, %s, %s, %s, %s
        from source s where s.id = %s
        returning id
        """,

         doc.extent_reason, json.dumps([{"id": t.id, "confidence": t.confidence} for t in doc.categories]),
         f.valid_from, f.valid_to, f.in_force, f.published_at, chash, version, f.source_id),

def _insert_chunks(conn: psycopg.Connection, document_id: int, doc: EnrichedDocument,
                   chunks: tuple[ChunkDraft, ...], embedder: Embedder) -> dict[str, int]:
    """Insert chunks; returns chunk_hash -> chunk id."""
    vectors = embedder.embed([c.text_en for c in chunks]) if chunks else []
    authority = conn.execute("select authority from document where id = %s", (document_id,)).fetchone()["authority"]
    ids: dict[str, int] = {}
    for chunk, vector in zip(chunks, vectors):
        chash = text_hash(chunk.text_en)
        row = conn.execute(
            """

        ).fetchone()
        ids[chash] = row["id"]
    return ids


def _diff(conn: psycopg.Connection, previous_id: int, new_ids: dict[str, int]) -> tuple[list[int], list[int], list[int]]:
    old = {r["chunk_hash"]: r["id"] for r in conn.execute(
        "select id, chunk_hash from chunk where document_id = %s", (previous_id,)).fetchall()}
    added = [cid for h, cid in new_ids.items() if h not in old]
    removed = [cid for h, cid in old.items() if h not in new_ids]
    # a "modified" chunk is one whose ordinal position exists in both versions but whose hash changed
    modified = added[: min(len(added), len(removed))]
    added = added[len(modified):]
    removed = removed[len(modified):]
    return added, removed, modified


def _log_change(conn: psycopg.Connection, run_id: str, document_id: int | None, previous_id: int | None,
                change_type: str, added=(), removed=(), modified=()) -> None:
    conn.execute(
        """
        insert into document_change (run_id, document_id, previous_document_id, change_type,
                                     chunks_added, chunks_removed, chunks_modified)
        values (%s, %s, %s, %s, %s, %s, %s)
        """,
        (run_id, document_id, previous_id, change_type, list(added), list(removed), list(modified)),
    )


def persist_document(conn: psycopg.Connection, doc: EnrichedDocument, chunks: tuple[ChunkDraft, ...],
                     embedder: Embedder, run_id: str) -> PersistOutcome:
    """Insert a new version when content changed; supersede the previous; log a chunk-level diff."""
    f = doc.fetched
    chash = content_hash(f)
    current = _current(conn, f.url)
    if current and current["content_hash"] == chash:
        touch_unchanged(conn, f.url)
        return PersistOutcome(document_id=current["id"], change_type=None)
    version = (current["version"] + 1) if current else 1
    document_id = _insert_document(conn, doc, chash, version)
    new_ids = _insert_chunks(conn, document_id, doc, chunks, embedder)
    save_known(conn, f.source_id, f.url, KnownUrl(etag=f.etag, last_modified=f.last_modified, content_hash=chash))
    if current is None:
        _log_change(conn, run_id, document_id, None, "added", added=new_ids.values())
        return PersistOutcome(document_id=document_id, change_type="added")
    conn.execute(
        "update document set status = 'superseded', superseded_by = %s, valid_to = coalesce(valid_to, %s) where id = %s",
        (document_id, f.valid_from, current["id"]),
    )
    conn.execute("update chunk set document_status = 'superseded' where document_id = %s", (current["id"],))
    added, removed, modified = _diff(conn, current["id"], new_ids)
    _log_change(conn, run_id, document_id, current["id"], "modified", added, removed, modified)
    return PersistOutcome(document_id=document_id, change_type="modified")


def touch_unchanged(conn: psycopg.Connection, url: str) -> None:
    conn.execute("update document set retrieved_at = now() where url = %s and status = 'current'", (url,))
    touch_known(conn, url)


def mark_withdrawn(conn: psycopg.Connection, url: str, run_id: str) -> bool:
    """Mark the current version withdrawn; keep the row so old citations resolve. Returns False if none."""
    current = _current(conn, url)
    if current is None:
        return False
    conn.execute("update document set status = 'withdrawn' where id = %s", (current["id"],))
    conn.execute("update chunk set document_status = 'withdrawn' where document_id = %s", (current["id"],))
    forget_known(conn, url)
    _log_change(conn, run_id, current["id"], None, "withdrawn")
    return True
```

Note on `valid_to`: for versioned law the new consolidation's `valid_from` closes the old one; for undated pages `valid_from` is `None`, so `coalesce(valid_to, NULL)` leaves it null and the `as_of` branch in `search.py` may then match several historical versions of an undated page. That is acceptable for this plan; Plan 3 adds a `distinct on (url)` if it shows up in evals.

```python
# src/swiss_grounding/ingest/pipeline.py
import logging
from dataclasses import dataclass, replace

import psycopg

from swiss_grounding.ingest.chunking import ChunkLimits, chunk_sections
from swiss_grounding.ingest.enrich import enrich
from swiss_grounding.ingest.fetch_state import load_known
from swiss_grounding.ingest.fetchers.base import Fetcher
from swiss_grounding.ingest.llm import LLM
from swiss_grounding.ingest.models import FetchError
from swiss_grounding.ingest.persist import mark_withdrawn, persist_document, touch_unchanged
from swiss_grounding.locations.tree import LocationTree
from swiss_grounding.registry.categories import CATEGORIES
from swiss_grounding.registry.models import Source
from swiss_grounding.retrieval.embedder import Embedder

log = logging.getLogger(__name__)


@dataclass(frozen=True)
class IngestReport:
    source_id: str
    run_id: str
    added: int = 0
    modified: int = 0
    unchanged: int = 0
    withdrawn: int = 0
    failed: int = 0
    errors: tuple[FetchError, ...] = ()

    def bump(self, field: str, error: FetchError | None = None) -> "IngestReport":
        errors = self.errors + ((error,) if error else ())
        return replace(self, **{field: getattr(self, field) + 1, "errors": errors})


def run_ingest(conn: psycopg.Connection, source: Source, fetcher: Fetcher, llm: LLM, embedder: Embedder,
               tree: LocationTree, run_id: str, limits: ChunkLimits = ChunkLimits()) -> IngestReport:
    """Run one source through fetch → enrich → chunk → persist. Failures are per document."""
    report = IngestReport(source_id=source.id, run_id=run_id)
    known = load_known(conn, source.id)
    result = fetcher.fetch(source, known)
    for error in result.errors:
        log.warning("fetch failed source=%s url=%s reason=%s", source.id, error.url, error.reason)
        report = report.bump("failed", error)
    for url in result.unchanged_urls:
        touch_unchanged(conn, url)
        report = report.bump("unchanged")
    for url in result.gone_urls:
        if mark_withdrawn(conn, url, run_id):
            report = report.bump("withdrawn")
    for fetched in result.documents:
        try:
            enriched = enrich(fetched, source, tree, llm, CATEGORIES)
            chunks = chunk_sections(fetched.sections, enriched.sections_en, limits)
            outcome = persist_document(conn, enriched, chunks, embedder, run_id)
        except Exception as exc:  # noqa: BLE001 - per-document isolation is the contract
            log.exception("ingest failed source=%s url=%s", source.id, fetched.url)
            report = report.bump("failed", FetchError(url=fetched.url, reason=str(exc)))
            continue
        if outcome.change_type is None:
            report = report.bump("unchanged")
        else:
            report = report.bump(outcome.change_type)
    log.info("ingest done %s", report)
    return report
```

Note: a document that fails enrichment is not written at all (no `pending_enrichment` row), which is the simplest correct behavior: the next run retries it because it is absent from `fetch_state`. The `pending_enrichment` status stays reserved in the schema for a future partial-write path.

```python
from urllib.parse import urlparse

from swiss_grounding.locations.seed import CANTONS

# exact host (without leading www.) -> extent id
CITY_DOMAINS: dict[str, str] = {
    "stadt-zuerich.ch": "CH-ZH-261",
}
FEDERAL_DOMAINS: tuple[str, ...] = ("ch.ch",)
FEDERAL_SUFFIXES: tuple[str, ...] = (".admin.ch",)
CANTON_DOMAINS: dict[str, str] = {f"{abbr.lower()}.ch": f"CH-{abbr}" for _, abbr, *_ in CANTONS}
SWISS_TLDS: tuple[str, ...] = (".ch", ".swiss")


def _host(url: str) -> str:
    host = urlparse(url).netloc.lower()
    return host[4:] if host.startswith("www.") else host


def extent_for_domain(url: str) -> str | None:
    host = _host(url)
    if host in CITY_DOMAINS:
        return CITY_DOMAINS[host]
    if host in CANTON_DOMAINS:
        return CANTON_DOMAINS[host]
    if host in FEDERAL_DOMAINS or host.endswith(FEDERAL_SUFFIXES):
        return "CH"
    return None


def classify_domain(url: str) -> str:
    """'official' for known government domains, 'swiss' for .ch/.swiss, else 'other'."""
    if extent_for_domain(url) is not None:
        return "official"
    return "swiss" if _host(url).endswith(SWISS_TLDS) else "other"
```

```python
# src/swiss_grounding/registry/suggestions.py
import re
from dataclasses import dataclass
from datetime import datetime
from urllib.parse import urlparse

import httpx
import psycopg

from swiss_grounding.registry.models import LAW_AUTHORITIES, Authority, FetchStrategy, Source
from swiss_grounding.registry.official_domains import classify_domain, extent_for_domain
from swiss_grounding.registry.repository import upsert_sources

SOURCE_ID_RE = re.compile(r"[^a-z0-9]+")


@dataclass(frozen=True)
class Suggestion:
    id: int
    url: str
    reason: str
    domain_class: str
    status: str
    proposed_extent_id: str | None
    proposed_authority: str | None
    proposed_categories: tuple[str, ...]
    suggested_by: str | None
    suggested_at: datetime
    review_note: str | None
    source_id: str | None


def _row_to_suggestion(r: dict) -> Suggestion:
    return Suggestion(
        id=r["id"], url=r["url"], reason=r["reason"], domain_class=r["domain_class"], status=r["status"],
        proposed_extent_id=r["proposed_extent_id"], proposed_authority=r["proposed_authority"],
        proposed_categories=tuple(r["proposed_categories"]), suggested_by=r["suggested_by"],
        suggested_at=r["suggested_at"], review_note=r["review_note"], source_id=r["source_id"],
    )


def _validate_url(url: str) -> None:
    parsed = urlparse(url)
    if parsed.scheme not in ("http", "https") or not parsed.netloc:
        raise ValueError(f"URL is not a valid http(s) URL: {url!r}")


def _check_reachable(http: httpx.Client, url: str) -> None:
    try:
        response = http.head(url, follow_redirects=True)
        if response.status_code >= 400:
            response = http.get(url, follow_redirects=True)
    except httpx.HTTPError as exc:
        raise ValueError(f"URL is not reachable: {exc}") from exc
    if response.status_code >= 400:
        raise ValueError(f"URL is not reachable (HTTP {response.status_code})")


def _check_not_known(conn: psycopg.Connection, url: str) -> None:
    host = urlparse(url).netloc.lower().removeprefix("www.")
    existing = conn.execute(
        "select id from source where lower(replace(base_url, 'www.', '')) like %s", (f"%{host}%",)
    ).fetchone()
    if existing:
        raise ValueError(f"domain already covered by source {existing['id']!r}")
    if conn.execute("select 1 from source_suggestion where url = %s", (url,)).fetchone():
        raise ValueError("URL already suggested")


def _source_id_for(url: str) -> str:
    host = urlparse(url).netloc.lower().removeprefix("www.")
    return SOURCE_ID_RE.sub("_", host).strip("_")


def _to_source(url: str, extent_id: str, authority: Authority, categories: tuple[str, ...], origin: str) -> Source:
    parsed = urlparse(url)
    return Source(
        id=_source_id_for(url), name=parsed.netloc, base_url=f"{parsed.scheme}://{parsed.netloc}",
        default_extent_id=extent_id, authority=authority, categories=categories,
        fetch_strategy=FetchStrategy.HTML_CRAWL,
        fetch_config={"path_allowlist": [parsed.path or "/"], "start_paths": [parsed.path or "/"]},
        allow_narrowing=authority not in LAW_AUTHORITIES, origin=origin,
    )


def submit_suggestion(conn: psycopg.Connection, http: httpx.Client, url: str, reason: str, suggested_by: str | None,
                      proposed_extent_id: str | None = None, proposed_authority: str | None = None,
                      proposed_categories: tuple[str, ...] = ()) -> Suggestion:
    """Validate and record a suggestion; official domains are approved immediately as official_info."""
    _validate_url(url)
    if proposed_authority and Authority(proposed_authority) in LAW_AUTHORITIES:
        raise ValueError("law authority levels can only be assigned by a human reviewer")
    _check_not_known(conn, url)
    _check_reachable(http, url)
    domain_class = classify_domain(url)
    row = conn.execute(
        """
        insert into source_suggestion (url, reason, proposed_extent_id, proposed_authority, proposed_categories,
                                       domain_class, suggested_by)
        values (%s, %s, %s, %s, %s, %s, %s) returning *
        """,
        (url, reason, proposed_extent_id, proposed_authority, list(proposed_categories), domain_class, suggested_by),
    ).fetchone()
    if domain_class != "official":
        return _row_to_suggestion(row)
    source = _to_source(url, extent_for_domain(url), Authority.OFFICIAL_INFO, proposed_categories, "auto_approved")
    upsert_sources(conn, (source,))
    row = conn.execute(
        "update source_suggestion set status = 'approved', reviewed_by = 'auto', reviewed_at = now(), "
        "review_note = 'official domain', source_id = %s where id = %s returning *",
        (source.id, row["id"]),
    ).fetchone()
    return _row_to_suggestion(row)


def review_suggestion(conn: psycopg.Connection, suggestion_id: int, approve: bool, reviewer: str, note: str,
                      extent_id: str | None = None, authority: str | None = None,
                      categories: tuple[str, ...] = ()) -> Suggestion:
    row = conn.execute("select * from source_suggestion where id = %s", (suggestion_id,)).fetchone()
    if row is None:
        raise ValueError(f"suggestion {suggestion_id} not found")
    if row["status"] != "pending":
        raise ValueError(f"suggestion {suggestion_id} is already {row['status']}")
    source_id = None
    if approve:
        chosen_extent = extent_id or row["proposed_extent_id"] or extent_for_domain(row["url"])
        if not chosen_extent:
            raise ValueError("an extent_id is required to approve a non-official source")
        chosen_authority = Authority(authority or row["proposed_authority"] or Authority.INSTITUTIONAL)
        source = _to_source(row["url"], chosen_extent, chosen_authority,
                            categories or tuple(row["proposed_categories"]), "reviewed")
        upsert_sources(conn, (source,))
        source_id = source.id
    updated = conn.execute(
        "update source_suggestion set status = %s, reviewed_by = %s, reviewed_at = now(), review_note = %s, "
        "source_id = %s where id = %s returning *",
        ("approved" if approve else "rejected", reviewer, note, source_id, suggestion_id),
    ).fetchone()
    return _row_to_suggestion(updated)


def list_suggestions(conn: psycopg.Connection, status: str | None = None) -> tuple[Suggestion, ...]:
    if status:
        rows = conn.execute("select * from source_suggestion where status = %s order by id", (status,)).fetchall()
    else:
        rows = conn.execute("select * from source_suggestion order by id").fetchall()
    return tuple(_row_to_suggestion(r) for r in rows)
```

```python
sources_app = typer.Typer(help="Source registry and suggestions.")

@app.command("ingest")
def ingest_cmd(
    source_id: str | None = typer.Option(None, "--source", help="source id; omit with --all"),
    all_sources: bool = typer.Option(False, "--all"),
) -> None:
    """Fetch, enrich, chunk, embed and persist one source or all sources. Exit 1 if a source is unknown."""
    if not source_id and not all_sources:
        typer.echo("pass --source <id> or --all", err=True)
        raise typer.Exit(code=1)
    settings = load_settings()
    embedder = build_embedder(settings.embedding_backend, settings.embedding_model, settings.embedding_dim)
    llm = build_llm(settings.llm_backend, settings.llm_model, settings.anthropic_api_key)
    run_id = uuid.uuid4().hex[:12]
    with connect(settings.database_url) as conn:
        tree = load_tree(conn)
        sources = load_sources(conn)
        selected = sources if all_sources else tuple(s for s in sources if s.id == source_id)
        if not selected:
            typer.echo(f"unknown source: {source_id}", err=True)
            raise typer.Exit(code=1)
        for source in selected:
            fetcher = build_fetcher(source.fetch_strategy, settings)
            report = run_ingest(conn, source, fetcher, llm, embedder, tree, run_id=run_id)
            typer.echo(
                f"{source.id}: added={report.added} modified={report.modified} unchanged={report.unchanged} "
                f"withdrawn={report.withdrawn} failed={report.failed} run={run_id}"
            )
            for error in report.errors:
                typer.echo(f"  ! {error.url}: {error.reason}", err=True)
```
